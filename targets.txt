# targets.txt — список адресов для команды `zapret test`.
#
# Формат (тот же, что у Flowseal):
#   Имя = "https://хост"   -> для адреса выполняются проверки TLS 1.2, TLS 1.3,
#                             HTTP/1.1, загрузка первых 64 КБ и (если curl
#                             умеет HTTP/3) QUIC
#   Имя = "PING:1.2.3.4"   -> строки PING пропускаются: ping в тесте не нужен
#
# Имя — одно слово из латинских букв, цифр и подчёркивания; оно же показывается
# в отчёте. Строки, начинающиеся с #, игнорируются.
# Можно добавлять свои строки и удалять лишние: больше целей — дольше проверка.
# Файл устанавливается в /opt/zapret-linux и при обновлении заменяется;
# свои цели удобнее держать в копии и указывать её через TARGETS_FILE.
